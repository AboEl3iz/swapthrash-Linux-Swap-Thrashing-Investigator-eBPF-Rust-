4/// Options to optionally be provided when attaching to a tracepoint.
5#[derive(Clone, Debug, Default)]
6#[doc(alias = "bpf_tracepoint_opts")]
7pub struct TracepointOpts {
8    /// Custom user-provided value accessible through `bpf_get_attach_cookie`.
9    pub cookie: u64,
10    #[doc(hidden)]
11    pub _non_exhaustive: (),
12}
13
14impl From<TracepointOpts> for libbpf_sys::bpf_tracepoint_opts {
15    fn from(opts: TracepointOpts) -> Self {
16        let TracepointOpts {
17            cookie,
18            _non_exhaustive,
19        } = opts;
20
21        #[allow(clippy::needless_update)]
22        Self {
23            sz: size_of::<Self>() as _,
24            bpf_cookie: cookie,
25            // bpf_tracepoint_opts might have padding fields on some platform
26            ..Default::default()
27        }
28    }
29}
